cmake_minimum_required(VERSION 3.24)
project(SkyrimWorldObserver VERSION 0.2.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_SKIP_INSTALL_RULES ON)
cmake_path(IS_PREFIX CMAKE_SOURCE_DIR "${CMAKE_BINARY_DIR}" NORMALIZE BUILD_INSIDE_REPO)
if(BUILD_INSIDE_REPO)
  message(FATAL_ERROR "Keep dependencies and build outputs outside this repository")
endif()
include(FetchContent)
file(READ "${CMAKE_SOURCE_DIR}/dependencies.json" OBSERVER_PINS)
function(observer_dependency target pin)
  string(JSON archive GET "${OBSERVER_PINS}" "${pin}" archiveUrl)
  string(JSON digest GET "${OBSERVER_PINS}" "${pin}" archiveSha256)
  FetchContent_Declare(${target} URL "${archive}" URL_HASH "SHA256=${digest}" ${ARGN})
endfunction()
set(OBSERVER_DEPENDENCY_CACHE "" CACHE PATH "Verified external _deps with CommonLibSSE-NG3.7.0 Release libraries")
option(OBSERVER_BUILD_PLUGIN "Build SKSEVR DLL" ON)
if(OBSERVER_DEPENDENCY_CACHE)
  set(JSON_INCLUDE "${OBSERVER_DEPENDENCY_CACHE}/nlohmann_json-src/include")
else()
  observer_dependency(nlohmann_json nlohmann_json)
  FetchContent_MakeAvailable(nlohmann_json)
  set(JSON_INCLUDE "${nlohmann_json_SOURCE_DIR}/include")
endif()
add_library(observer_contract INTERFACE)
target_include_directories(observer_contract INTERFACE src "${JSON_INCLUDE}")
enable_testing()
add_executable(observer_contract_tests tests/contract.cpp)
target_link_libraries(observer_contract_tests PRIVATE observer_contract)
add_test(NAME observer_contract COMMAND observer_contract_tests)
if(NOT OBSERVER_BUILD_PLUGIN)
  return()
endif()
if(NOT WIN32)
  message(FATAL_ERROR "SKSEVR plugin requires Windows x64")
endif()
if(OBSERVER_DEPENDENCY_CACHE)
  add_library(CommonLibSSE STATIC IMPORTED)
  set_target_properties(CommonLibSSE PROPERTIES IMPORTED_LOCATION "${OBSERVER_DEPENDENCY_CACHE}/commonlibsse-build/Release/CommonLibSSE.lib")
  target_include_directories(CommonLibSSE INTERFACE
    "${OBSERVER_DEPENDENCY_CACHE}/commonlibsse-src/include"
    "${OBSERVER_DEPENDENCY_CACHE}/fmt-src/include"
    "${OBSERVER_DEPENDENCY_CACHE}/spdlog-src/include"
    "${OBSERVER_DEPENDENCY_CACHE}/rapidcsv-src/src")
  target_compile_definitions(CommonLibSSE INTERFACE ENABLE_SKYRIM_SE=1 ENABLE_SKYRIM_AE=1 ENABLE_SKYRIM_VR=1 HAS_SKYRIM_MULTI_TARGETING=1 SPDLOG_COMPILED_LIB SPDLOG_FMT_EXTERNAL)
  target_link_libraries(CommonLibSSE INTERFACE "${OBSERVER_DEPENDENCY_CACHE}/fmt-build/Release/fmt.lib" "${OBSERVER_DEPENDENCY_CACHE}/spdlog-build/Release/spdlog.lib" Version.lib)
else()
  set(FMT_INSTALL OFF CACHE BOOL "" FORCE)
  set(SPDLOG_INSTALL OFF CACHE BOOL "" FORCE)
  set(SPDLOG_FMT_EXTERNAL ON CACHE BOOL "" FORCE)
  set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
  observer_dependency(fmt fmt OVERRIDE_FIND_PACKAGE)
  observer_dependency(spdlog spdlog OVERRIDE_FIND_PACKAGE)
  observer_dependency(rapidcsv rapidcsv SOURCE_SUBDIR src)
  FetchContent_MakeAvailable(fmt spdlog rapidcsv)
  set(RAPIDCSV_INCLUDE_DIRS "${rapidcsv_SOURCE_DIR}/src" CACHE PATH "" FORCE)
  observer_dependency(CommonLibSSE CommonLibSSE-NG)
  FetchContent_MakeAvailable(CommonLibSSE)
  # CommonLib's public Relocation header includes rapidcsv, but 3.7.0
  # propagates that include only privately to its own compilation.
  target_include_directories(CommonLibSSE INTERFACE "${rapidcsv_SOURCE_DIR}/src")
endif()
set(DEVBENCH_API_DIR "${CMAKE_BINARY_DIR}/external/devbench-api")
foreach(file DevBenchAPI.h DevBenchAPI.cpp DevBenchAPI.LICENSE.txt)
  if(NOT EXISTS "${DEVBENCH_API_DIR}/${file}")
    message(FATAL_ERROR "Run tools/build.py to fetch and verify the pinned DevBench integration ABI")
  endif()
endforeach()
add_library(SkyrimWorldObserver SHARED src/plugin.cpp src/Physics.cpp "${DEVBENCH_API_DIR}/DevBenchAPI.cpp")
target_include_directories(SkyrimWorldObserver PRIVATE "${DEVBENCH_API_DIR}")
target_link_libraries(SkyrimWorldObserver PRIVATE CommonLibSSE observer_contract bcrypt)
target_compile_definitions(SkyrimWorldObserver PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(SkyrimWorldObserver PRIVATE /W4 /EHsc /utf-8 /wd4200 /wd4201)
set_target_properties(SkyrimWorldObserver PROPERTIES PREFIX "" SUFFIX ".dll")
