option(OBSERVER_BUILD_SE104 "Build exact Skyrim SE1.7.104 subset with a verified modern SDK" OFF)
if(OBSERVER_BUILD_SE104)
  if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "SE104 plugin requires Windows x64")
  endif()
  foreach(input OBSERVER_SE_SDK_ROOT OBSERVER_SE_SDK_LIBRARY OBSERVER_SE_INCLUDE_DIRS OBSERVER_SE_LINK_LIBRARIES DEVBENCH_API_DIR)
    if(NOT DEFINED ${input} OR "${${input}}" STREQUAL "")
      message(FATAL_ERROR "Explicit verified modern build input required: ${input}")
    endif()
  endforeach()
  if(NOT EXISTS "${OBSERVER_SE_SDK_LIBRARY}")
    message(FATAL_ERROR "Verified source-built modern SDK static library missing")
  endif()
  foreach(api DevBenchAPI.h DevBenchAPI.cpp DevBenchAPI.LICENSE.txt)
    if(NOT EXISTS "${DEVBENCH_API_DIR}/${api}")
      message(FATAL_ERROR "Pinned external DevBench ABI missing: ${api}")
    endif()
  endforeach()
  add_library(observer_modern_sdk STATIC IMPORTED)
  set_target_properties(observer_modern_sdk PROPERTIES IMPORTED_LOCATION "${OBSERVER_SE_SDK_LIBRARY}")
  target_include_directories(observer_modern_sdk INTERFACE "${OBSERVER_SE_SDK_ROOT}/include"
    "${OBSERVER_SE_SDK_ROOT}/extern/openvr/headers" ${OBSERVER_SE_INCLUDE_DIRS})
  target_compile_definitions(observer_modern_sdk INTERFACE ENABLE_SKYRIM_SE=1 ENABLE_SKYRIM_AE=1 ENABLE_SKYRIM_VR=1
    HAS_SKYRIM_MULTI_TARGETING=1 SKSE_SUPPORT_PATCH_SAFETY=1 SKSE_SUPPORT_XBYAK=1 REX_OPTION_INI=1
    SPDLOG_COMPILED_LIB SPDLOG_USE_STD_FORMAT SPDLOG_WCHAR_TO_UTF8_SUPPORT SI_CONVERT_WIN32)
  target_link_libraries(observer_modern_sdk INTERFACE ${OBSERVER_SE_LINK_LIBRARIES}
    advapi32 bcrypt d3d11 d3dcompiler dbghelp dxgi ole32 shell32 user32 version)
  add_library(SkyrimWorldObserver SHARED src/plugin_se.cpp "${DEVBENCH_API_DIR}/DevBenchAPI.cpp")
  target_include_directories(SkyrimWorldObserver PRIVATE src "${DEVBENCH_API_DIR}")
  target_link_libraries(SkyrimWorldObserver PRIVATE observer_modern_sdk)
  target_compile_definitions(SkyrimWorldObserver PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(SkyrimWorldObserver PRIVATE /W4 /EHsc /utf-8 /permissive- /Zc:preprocessor /bigobj /wd4200 /wd4201)
  target_precompile_headers(SkyrimWorldObserver PRIVATE "${OBSERVER_SE_SDK_ROOT}/include/SKSE/Impl/PCH.h")
  set_property(TARGET SkyrimWorldObserver PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
  set_target_properties(SkyrimWorldObserver PROPERTIES PREFIX "" SUFFIX ".dll" COMPILE_PDB_NAME "SkyrimWorldObserver-compile")
  target_compile_options(SkyrimWorldObserver PRIVATE /Zi)
  target_link_options(SkyrimWorldObserver PRIVATE /DEBUG)
  enable_testing()
  add_executable(observer_contract_tests tests/contract.cpp)
  target_include_directories(observer_contract_tests PRIVATE src ${OBSERVER_SE_INCLUDE_DIRS})
  add_test(NAME observer_contract COMMAND observer_contract_tests)
  add_executable(observer_se_contract_tests tests/se_contract.cpp)
  target_include_directories(observer_se_contract_tests PRIVATE src ${OBSERVER_SE_INCLUDE_DIRS})
  add_test(NAME observer_se_contract COMMAND observer_se_contract_tests)
  return()
endif()
include(FetchContent)
file(READ "${CMAKE_SOURCE_DIR}/dependencies.json" OBSERVER_PINS)
function(observer_dependency target pin)
  string(JSON archive GET "${OBSERVER_PINS}" "${pin}" archiveUrl)
  string(JSON digest GET "${OBSERVER_PINS}" "${pin}" archiveSha256)
  FetchContent_Declare(${target} URL "${archive}" URL_HASH "SHA256=${digest}" ${ARGN})
endfunction()

if(OBSERVER_DEPENDENCY_CACHE)
  add_library(CommonLibSSE STATIC IMPORTED)
  set_target_properties(CommonLibSSE PROPERTIES IMPORTED_LOCATION "${OBSERVER_DEPENDENCY_CACHE}/commonlibsse-build/Release/CommonLibSSE.lib")
  target_include_directories(CommonLibSSE INTERFACE
    "${OBSERVER_DEPENDENCY_CACHE}/commonlibsse-src/include"
    "${OBSERVER_DEPENDENCY_CACHE}/fmt-src/include"
    "${OBSERVER_DEPENDENCY_CACHE}/spdlog-src/include"
    "${OBSERVER_DEPENDENCY_CACHE}/rapidcsv-src/src")
  target_compile_definitions(CommonLibSSE INTERFACE ENABLE_SKYRIM_SE=1 ENABLE_SKYRIM_AE=1 ENABLE_SKYRIM_VR=1 HAS_SKYRIM_MULTI_TARGETING=1 SPDLOG_COMPILED_LIB SPDLOG_FMT_EXTERNAL)
  target_link_libraries(CommonLibSSE INTERFACE "${OBSERVER_DEPENDENCY_CACHE}/fmt-build/Release/fmt.lib" "${OBSERVER_DEPENDENCY_CACHE}/spdlog-build/Release/spdlog.lib" Version.lib)
else()
  set(FMT_INSTALL OFF CACHE BOOL "" FORCE)
  set(SPDLOG_INSTALL OFF CACHE BOOL "" FORCE)
  set(SPDLOG_FMT_EXTERNAL ON CACHE BOOL "" FORCE)
  set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
  observer_dependency(fmt fmt OVERRIDE_FIND_PACKAGE)
  observer_dependency(spdlog spdlog OVERRIDE_FIND_PACKAGE)
  observer_dependency(rapidcsv rapidcsv SOURCE_SUBDIR src)
  FetchContent_MakeAvailable(fmt spdlog rapidcsv)
  set(RAPIDCSV_INCLUDE_DIRS "${rapidcsv_SOURCE_DIR}/src" CACHE PATH "" FORCE)
  observer_dependency(CommonLibSSE CommonLibSSE-NG)
  FetchContent_MakeAvailable(CommonLibSSE)
  # CommonLib's public Relocation header includes rapidcsv, but 3.7.0
  # propagates that include only privately to its own compilation.
  target_include_directories(CommonLibSSE INTERFACE "${rapidcsv_SOURCE_DIR}/src")
